-- PART 2 — Five required queries (run in MySQL Workbench after ETL load)
-- =============================================================================
USE sales_db;

-- 1) Total sales by region and category
SELECT
    t.region,
    p.category,
    ROUND(SUM(t.total_value), 2) AS total_sales
FROM transactions t
JOIN products p ON t.product_id = p.product_id
GROUP BY t.region, p.category
ORDER BY total_sales DESC;

-- 2) Top 5 products by revenue
SELECT
    p.product_id,
    p.product_name,
    ROUND(SUM(t.total_value), 2) AS total_revenue
FROM transactions t
JOIN products p ON t.product_id = p.product_id
GROUP BY p.product_id, p.product_name
ORDER BY total_revenue DESC
LIMIT 5;

-- 3) Monthly sales trend
SELECT
    DATE_FORMAT(t.transaction_date, '%Y-%m') AS sales_month,
    ROUND(SUM(t.total_value), 2) AS monthly_sales,
    COUNT(*) AS transaction_count
FROM transactions t
GROUP BY DATE_FORMAT(t.transaction_date, '%Y-%m')
ORDER BY sales_month;

-- 4) Average discount % per region
SELECT
    t.region,
    ROUND(AVG(t.discount_pct) * 100, 2) AS avg_discount_percent
FROM transactions t
GROUP BY t.region
ORDER BY avg_discount_percent DESC;

-- 5) Count of transactions with total_value > 1000
SELECT COUNT(*) AS high_value_transaction_count
FROM transactions
WHERE total_value > 1000;
